name: CalloutBox

export const CalloutBox = ({ type = 'info', title, children }) => {
  let style = '';
  let icon = '';
  
  if (type === 'info') {
    style = 'border-blue-300 dark:border-blue-500 bg-blue-50 dark:bg-blue-900 text-blue-900 dark:text-blue-100';
    icon = 'fa-info-circle';
  } else if (type === 'success') {
    style = 'border-green-300 dark:border-green-500 bg-green-50 dark:bg-green-900 text-green-900 dark:text-green-100';
    icon = 'fa-check-circle';
  } else if (type === 'warning') {
    style = 'border-yellow-300 dark:border-yellow-500 bg-yellow-50 dark:bg-yellow-900 text-yellow-900 dark:text-yellow-100';
    icon = 'fa-exclamation-triangle';
  } else if (type === 'danger') {
    style = 'border-red-300 dark:border-red-500 bg-red-50 dark:bg-red-900 text-red-900 dark:text-red-100';
    icon = 'fa-times-circle';
  }

  return (
    <div className="flex justify-center">
      <div className={`rounded-md p-6 m-4 max-w-lg shadow-md border ${style}`}>
        <p className="text-lg font-bold mb-2">
          <i className={`fa-solid ${icon} mr-2`} />
          {title}
        </p>
        <div className="text-sm leading-relaxed">{children}</div>
      </div>
    </div>
  );
};

<CalloutBox type="info" title="Pet Fact">
  Dogs can hear frequencies as high as 65,000 Hz, far higher than humans.
</CalloutBox>
